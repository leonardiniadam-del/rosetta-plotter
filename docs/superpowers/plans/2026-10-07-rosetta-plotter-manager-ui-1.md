```python
def _rotation(rotation) -> float:
    """Degrees in [0, 360). ValueError for anything that isn't a finite number."""
    if isinstance(rotation, bool) or not isinstance(rotation, (int, float)) or not math.isfinite(rotation):
        raise ValueError(f"Rotation must be a finite number of degrees: {rotation!r}")
    return float(rotation) % 360.0


def _cos_sin(turn: float) -> tuple[float, float]:
    """Exact for quarter turns, so placements at 0/90/180/270 land on exact numbers."""
    exact = {0.0: (1.0, 0.0), 90.0: (0.0, 1.0), 180.0: (-1.0, 0.0), 270.0: (0.0, -1.0)}.get(turn)
    if exact is not None:
        return exact
    radians = math.radians(turn)
    return math.cos(radians), math.sin(radians)


def rotated_size(width: float, height: float, rotation) -> tuple[float, float]:
    """Size of the upright box around a width x height frame turned by rotation degrees."""
    turn = _rotation(rotation)
    if turn in (90.0, 270.0):
        return (height, width)
    if turn in (0.0, 180.0):
        return (width, height)
    cos, sin = (abs(value) for value in _cos_sin(turn))
    return (width * cos + height * sin, width * sin + height * cos)


def place_point(
    point: Point,
    art_width: float,
    art_height: float,
    x: float,
    y: float,
    scale: float,
    rotation,
    origin: Point,
    scale_y: float | None = None,
) -> Point:
    """Artwork point -> machine point.

    The artwork is scaled on its own axes (scale across its width, scale_y down its height; None means
    the same as scale), turned clockwise by rotation degrees, and the top-left of its upright box lands
    at origin + (x, y). origin is the material's corner nearest Home.
    """
    scale_x, scale_y = scale, scale if scale_y is None else scale_y
    cos, sin = _cos_sin(_rotation(rotation))
    # The linear part: turn(scale(point)). A frame corner lands at (a*u + c*v, b*u + d*v).
    a, b, c, d = cos * scale_x, sin * scale_x, -sin * scale_y, cos * scale_y
    xs = (0.0, a * art_width, a * art_width + c * art_height, c * art_height)
    ys = (0.0, b * art_width, b * art_width + d * art_height, d * art_height)
    px, py = point
    return (
        origin[0] + x + a * px + c * py - min(xs),
        origin[1] + y + b * px + d * py - min(ys),
    )
```

    if not (math.isfinite(placement.x) and math.isfinite(placement.y)):

"""What the student is putting together (manager UI spec §7): the Jobs list of imported files, the
pieces placed from them on the paper, and the pen colors to leave out.

The plotter's model and pen settings live on Machine; the plot itself in PlotController.
Methods here are called from several bridge threads; each change is one list or attribute update.
"""
import dataclasses
import itertools
import math
from dataclasses import dataclass
from pathlib import Path

from rosetta.compose import DEFAULT_MARGIN_MM, Job, Material, PlacedPiece, Placement, compose_job
from rosetta.errors import ImportProblem, UserProblem
from rosetta.geometry import rotated_size
from rosetta.loader import Artwork, load_artwork
from rosetta.models import MachineModel
from rosetta.pdf_import import page_count

DUPLICATE_OFFSET_MM = 10.0
MAX_SCALE = 100.0
GONE = "That drawing isn't on the paper anymore."
JOB_GONE = "That file isn't in the Jobs list anymore."


@dataclass
class QueuedJob:
    """An imported file waiting in the Jobs list. It can be placed on the paper any number of times."""

    id: str
    path: str
    page: int  # 0-based
    artwork: Artwork


@dataclass
class Piece:
    """One placed copy of a job."""

    id: str
    job_id: str
    artwork: Artwork
    placement: Placement


def pages_in(path) -> int:
    """How many pages a file has: PDF and AI files can have several; everything else has one."""
    path = Path(path)
    if not path.is_file():
        raise ImportProblem(f"The file {path.name} can't be found.")
    if path.suffix.lower() in (".pdf", ".ai"):
        return page_count(path)
    return 1


class Session:
    def __init__(self, material: Material, margin_mm: float = DEFAULT_MARGIN_MM):
        self.material = material
        self.margin_mm = margin_mm
        self.skip_white = True
        self.jobs: list[QueuedJob] = []
        self.pieces: list[Piece] = []
        self.hidden_colors: set[str] = set()  # lowercase '#rrggbb' pens left out of the plot
        self._job_ids = itertools.count(1)
        self._piece_ids = itertools.count(1)

    # The Jobs list.

    def load_job(self, path, page: int = 0) -> QueuedJob:
        """Import a file into the Jobs list (slow for big files). Nothing is placed yet."""
        artwork = load_artwork(Path(path), page)
        job = QueuedJob(f"job-{next(self._job_ids)}", str(path), page, artwork)
        self.jobs.append(job)
        return job

    def get_job(self, job_id: str) -> QueuedJob:
        for job in self.jobs:
            if job.id == job_id:
                return job
        raise UserProblem(JOB_GONE)

    def remove_job(self, job_id: str) -> list[str]:
        """Take a file off the Jobs list, with every copy of it on the paper. Returns those pieces' ids."""
        job = self.get_job(job_id)
        removed = [piece.id for piece in self.pieces if piece.job_id == job_id]
        self.pieces = [piece for piece in self.pieces if piece.job_id != job_id]
        self.jobs.remove(job)
        return removed

    def place_job(self, job_id: str, x, y) -> Piece:
        """Put a copy of a job on the paper at full size, its top-left corner at (x, y) from the paper's corner."""
        job = self.get_job(job_id)
        placement = _checked(Placement(float(x), float(y), 1.0, 0.0))
        piece = Piece(f"piece-{next(self._piece_ids)}", job.id, job.artwork, placement)
        self.pieces.append(piece)
        return piece

    # Pieces on the paper.

    def get(self, piece_id: str) -> Piece:
        for piece in self.pieces:
            if piece.id == piece_id:
                return piece
        raise UserProblem(GONE)

    def place(self, piece_id: str, x, y, scale, rotation, scale_y=None) -> Piece:
        piece = self.get(piece_id)
        piece.placement = _checked(Placement(
            float(x), float(y), float(scale), float(rotation) % 360.0,
            None if scale_y is None else float(scale_y),
        ))
        return piece

    def duplicate(self, piece_id: str) -> Piece:
        original = self.get(piece_id)
        placement = dataclasses.replace(
            original.placement,
            x=original.placement.x + DUPLICATE_OFFSET_MM,
            y=original.placement.y + DUPLICATE_OFFSET_MM,
        )
        copy = Piece(f"piece-{next(self._piece_ids)}", original.job_id, original.artwork, placement)
        self.pieces.insert(self.pieces.index(original) + 1, copy)
        return copy

    def remove(self, piece_id: str) -> None:
        self.pieces.remove(self.get(piece_id))

    def clear(self) -> None:
        self.pieces.clear()

    def rotate_paper(self) -> None:
        """Turn the paper and everything on it 90° clockwise (manager UI spec §4.2).

        A point (px, py) on the sheet moves to (H - py, px), H being the sheet's height before the turn,
        so each piece's box (x, y, w, h) becomes (H - y - h, x, h, w) and its angle gains 90°.
        """
        old = self.material
        self.material = Material(old.x, old.y, old.height, old.width)
        for piece in self.pieces:
            placement, art = piece.placement, piece.artwork
            scale_y = placement.scale if placement.scale_y is None else placement.scale_y
            _width, height = rotated_size(art.width_mm * placement.scale, art.height_mm * scale_y, placement.rotation)
            piece.placement = dataclasses.replace(
                placement,
                x=old.height - placement.y - height,
                y=placement.x,
                rotation=(placement.rotation + 90.0) % 360.0,
            )

    # Pens.

    def set_color_visible(self, color: str, visible: bool) -> None:
        color = color.lower()
        if visible:
            self.hidden_colors.discard(color)
        else:
            self.hidden_colors.add(color)

    def compose(self, model: MachineModel, include_hidden: bool = False) -> Job:
        placed = [PlacedPiece(piece.artwork, piece.placement) for piece in list(self.pieces)]
        skip = frozenset() if include_hidden else frozenset(self.hidden_colors)
        return compose_job(model, self.material, placed, skip_white=self.skip_white, skip_colors=skip)


def _checked(placement: Placement) -> Placement:
    """Placements come from the page; refuse values that would put NaN or mirrored art in a plot."""
    if not (math.isfinite(placement.x) and math.isfinite(placement.y)):
        raise ValueError(f"Placement position must be finite: {placement}")
    for scale in (placement.scale, placement.scale if placement.scale_y is None else placement.scale_y):
        if not (math.isfinite(scale) and 0 < scale <= MAX_SCALE):
            raise ValueError(f"Placement scale must be above 0 and at most {MAX_SCALE:g}: {placement}")
    if not math.isfinite(placement.rotation):
        raise ValueError(f"Placement rotation must be a finite number of degrees: {placement}")
    return placement

/** Lengths: everything inside the app is millimetres; the page shows inches, millimetres or centimetres. */
export type Units = "in" | "mm" | "cm";

export const UNITS: readonly Units[] = ["in", "mm", "cm"];
export const UNIT_NAMES: Record<Units, string> = { in: "Inches", mm: "Millimeters", cm: "Centimeters" };
export const MM_PER: Record<Units, number> = { in: 25.4, mm: 1, cm: 10 };
const DECIMALS: Record<Units, number> = { in: 2, mm: 1, cm: 2 };

export function toUnits(mm: number, units: Units): number {
  return mm / MM_PER[units];
}

/** "3.25", or "3.25 in" with the unit; rounded for the units, never "-0.00". */
export function formatLength(mm: number, units: Units, withUnit = false): string {
  const digits = DECIMALS[units];
  const rounded = Number(toUnits(mm, units).toFixed(digits));
  const text = (rounded === 0 ? 0 : rounded).toFixed(digits);
  return withUnit ? `${text} ${units}` : text;
}

const LENGTH = /^\s*([+-]?(?:\d+(?:\.\d*)?|\.\d+))\s*(in|inch|inches|"|mm|cm)?\s*$/i;

/** A typed length in mm: "2", "2 in", '2"', "50mm", "5 cm". A bare number is in the display units. */
export function parseLength(text: string, units: Units): number | null {
  const match = LENGTH.exec(text);
  if (!match) return null;
  const value = Number(match[1]);
  if (!Number.isFinite(value)) return null;
  const suffix = (match[2] ?? "").toLowerCase();
  const unit: Units = suffix === "" ? units : suffix === "mm" ? "mm" : suffix === "cm" ? "cm" : "in";
  return value * MM_PER[unit];
}

const NUMBER = /^\s*([+-]?(?:\d+(?:\.\d*)?|\.\d+))\s*(%|°|deg)?\s*$/i;

/** "50" or "50%" → 0.5. Null if unreadable or not above zero. */
export function parsePercent(text: string): number | null {
  const match = NUMBER.exec(text);
  if (!match || (match[2] && match[2] !== "%")) return null;
  const value = Number(match[1]) / 100;
  return Number.isFinite(value) && value > 0 ? value : null;
}

/** "30", "30°" or "-90" → degrees in [0, 360). */
export function parseAngle(text: string): number | null {
  const match = NUMBER.exec(text);
  if (!match || match[2] === "%") return null;
  const value = Number(match[1]);
  return Number.isFinite(value) ? normalizeAngle(value) : null;
}

/** Degrees in [0, 360); never -0. */
export function normalizeAngle(degrees: number): number {
  const value = degrees % 360;
  if (value < 0) return value + 360;
  return value === 0 ? 0 : value;
}

export function formatPercent(scale: number): string {
  return `${Number((scale * 100).toFixed(1))}%`;
}

export function formatAngle(degrees: number): string {
  return `${normalizeAngle(Number(normalizeAngle(degrees).toFixed(1)))}°`;
}

/**
 * Where placed drawings land: the same math as rosetta.geometry.place_point, so the screen shows exactly
 * what will be plotted. Units are mm; y grows toward the student; angles turn clockwise on screen.
 * A placement scales the artwork on its own axes (scale across, scale_y down), turns it, and puts the
 * top-left of its upright box at (x, y) from the paper's corner nearest Home.
 */
import { normalizeAngle } from "./units";

export type Point = [number, number];
export interface Placement {
  x: number;
  y: number;
  scale: number;
  scale_y: number;
  rotation: number;
}
export interface Box {
  x: number;
  y: number;
  width: number;
  height: number;
}
/** An SVG matrix(a b c d e f): x' = a·x + c·y + e, y' = b·x + d·y + f. */
export type Matrix = [number, number, number, number, number, number];
export type Handle = "nw" | "ne" | "se" | "sw";

export const MIN_SIZE_MM = 0.5;

const QUARTER: Record<number, Point> = { 0: [1, 0], 90: [0, 1], 180: [-1, 0], 270: [0, -1] };

export function isQuarterTurn(rotation: number): boolean {
  return normalizeAngle(rotation) in QUARTER;
}

/** Exact for quarter turns, like Python's _cos_sin. */
function cosSin(rotation: number): Point {
  const turn = normalizeAngle(rotation);
  const exact = QUARTER[turn];
  if (exact) return exact;
  const radians = (turn * Math.PI) / 180;
  return [Math.cos(radians), Math.sin(radians)];
}

/** Size of the upright box around a width × height frame turned by rotation degrees. */
export function rotatedSize(width: number, height: number, rotation: number): Point {
  const [cos, sin] = cosSin(rotation).map(Math.abs) as Point;
  return [width * cos + height * sin, width * sin + height * cos];
}

/** Artwork mm → machine mm, as an SVG matrix. origin is the paper's corner nearest Home. */
export function pieceMatrix(artWidth: number, artHeight: number, placement: Placement, origin: Point): Matrix {
  const [cos, sin] = cosSin(placement.rotation);
  const a = cos * placement.scale;
  const b = sin * placement.scale;
  const c = -sin * placement.scale_y;
  const d = cos * placement.scale_y;
  const xs = [0, a * artWidth, a * artWidth + c * artHeight, c * artHeight];
  const ys = [0, b * artWidth, b * artWidth + d * artHeight, d * artHeight];
  return [a, b, c, d, origin[0] + placement.x - Math.min(...xs), origin[1] + placement.y - Math.min(...ys)];
}

export function applyMatrix(matrix: Matrix, point: Point): Point {
  const [a, b, c, d, e, f] = matrix;
  return [a * point[0] + c * point[1] + e, b * point[0] + d * point[1] + f];
}

export function placePoint(point: Point, artWidth: number, artHeight: number, placement: Placement, origin: Point): Point {
  return applyMatrix(pieceMatrix(artWidth, artHeight, placement, origin), point);
}

/** The drawing's upright box, from the paper's corner (like Placement's x, y). */
export function pieceBox(artWidth: number, artHeight: number, placement: Placement): Box {
  const [width, height] = rotatedSize(artWidth * placement.scale, artHeight * placement.scale_y, placement.rotation);
  return { x: placement.x, y: placement.y, width, height };
}

/** Turn a drawing to a new angle about its center. */
export function rotateAboutCenter(artWidth: number, artHeight: number, placement: Placement, rotation: number): Placement {
  const box = pieceBox(artWidth, artHeight, placement);
  const turned = { ...placement, rotation: normalizeAngle(rotation) };
  const next = pieceBox(artWidth, artHeight, turned);
  return { ...turned, x: box.x + (box.width - next.width) / 2, y: box.y + (box.height - next.height) / 2 };
}

/**
 * A typed W and/or H for the drawing's upright box, keeping its top-left.
 * Linked: both scales change by one factor (from W if given, else H).
 * Unlinked: stretches along the drawing's own sides, which needs a quarter turn (null otherwise).
 * Null too for sizes that aren't above zero.
 */
export function resizeBox(
  artWidth: number,
  artHeight: number,
  placement: Placement,
  size: { width?: number; height?: number },
  linked: boolean,
): Placement | null {
  const box = pieceBox(artWidth, artHeight, placement);
  if (linked) {
    const factor = size.width !== undefined ? size.width / box.width : size.height !== undefined ? size.height / box.height : 1;
    if (!(factor > 0) || !Number.isFinite(factor)) return null;
    return { ...placement, scale: placement.scale * factor, scale_y: placement.scale_y * factor };
  }
  if (!isQuarterTurn(placement.rotation)) return null;
  const turn = normalizeAngle(placement.rotation);
  const widthIsOwnWidth = turn === 0 || turn === 180;
  let { scale, scale_y } = placement;
  if (size.width !== undefined) {
    if (widthIsOwnWidth) scale = size.width / artWidth;
    else scale_y = size.width / artHeight;
  }
  if (size.height !== undefined) {
    if (widthIsOwnWidth) scale_y = size.height / artHeight;
    else scale = size.height / artWidth;
  }
  if (!(scale > 0 && scale_y > 0) || !Number.isFinite(scale) || !Number.isFinite(scale_y)) return null;
  return { ...placement, scale, scale_y };
}

/**
 * Dragging a corner handle to `pointer` (paper mm) with the opposite corner fixed. Stretches freely at a
 * quarter turn unless keepProportions (Shift); always keeps proportions at a free angle. Never smaller than
 * MIN_SIZE_MM on a side.
 */
export function scaleFromHandle(
  artWidth: number,
  artHeight: number,
  placement: Placement,
  handle: Handle,
  pointer: Point,
  keepProportions: boolean,
): Placement {
  const box = pieceBox(artWidth, artHeight, placement);
  const west = handle === "nw" || handle === "sw";
  const north = handle === "nw" || handle === "ne";
  const fixedX = west ? box.x + box.width : box.x;
  const fixedY = north ? box.y + box.height : box.y;
  const width = Math.max(MIN_SIZE_MM, Math.abs(pointer[0] - fixedX));
  const height = Math.max(MIN_SIZE_MM, Math.abs(pointer[1] - fixedY));
  let resized: Placement | null;
  if (keepProportions || !isQuarterTurn(placement.rotation)) {
    const factor = Math.max(width / box.width, height / box.height);
    resized = resizeBox(artWidth, artHeight, placement, { width: box.width * factor }, true);
  } else {
    resized = resizeBox(artWidth, artHeight, placement, { width, height }, false);
  }
  const next = resized ?? placement;
  const size = pieceBox(artWidth, artHeight, next);
  return { ...next, x: west ? fixedX - size.width : fixedX, y: north ? fixedY - size.height : fixedY };
}

/** The plate's view: screen px = machine mm × scale + (x, y). */
import type { Box, Point } from "./placement";

export interface Camera {
  scale: number;
  x: number;
  y: number;
}
export interface Size {
  width: number;
  height: number;
}

export const MIN_SCALE = 0.05;
export const MAX_SCALE = 200;

function clampScale(scale: number): number {
  return Math.min(MAX_SCALE, Math.max(MIN_SCALE, Number.isFinite(scale) ? scale : MAX_SCALE));
}

export function fitCamera(box: Box, view: Size, padding = 32): Camera {
  const scale = clampScale(
    Math.min((view.width - 2 * padding) / box.width, (view.height - 2 * padding) / box.height),
  );
  return {
    scale,
    x: (view.width - box.width * scale) / 2 - box.x * scale,
    y: (view.height - box.height * scale) / 2 - box.y * scale,
  };
}

export function toScreen(camera: Camera, point: Point): Point {
  return [point[0] * camera.scale + camera.x, point[1] * camera.scale + camera.y];
}

export function toMachine(camera: Camera, point: Point): Point {
  return [(point[0] - camera.x) / camera.scale, (point[1] - camera.y) / camera.scale];
}

/** Zoom by factor, keeping the machine point under `screen` where it is. */
export function zoomAt(camera: Camera, factor: number, screen: Point): Camera {
  const scale = clampScale(camera.scale * factor);
  const [mx, my] = toMachine(camera, screen);
  return { scale, x: screen[0] - mx * scale, y: screen[1] - my * scale };
}

export function panBy(camera: Camera, dx: number, dy: number): Camera {
  return { ...camera, x: camera.x + dx, y: camera.y + dy };
}

/** Pure helpers for the plate: hit testing, the marquee, handles, angles, paths and the grid. */
import type { Box, Handle, Point } from "../geometry/placement";
import { normalizeAngle } from "../geometry/units";

export interface IdBox {
  id: string;
  box: Box;
}

function contains(box: Box, [x, y]: Point): boolean {
  return x >= box.x && x <= box.x + box.width && y >= box.y && y <= box.y + box.height;
}

function overlaps(a: Box, b: Box): boolean {
  return a.x <= b.x + b.width && b.x <= a.x + a.width && a.y <= b.y + b.height && b.y <= a.y + a.height;
}

/** The topmost (last drawn) box containing the point. */
export function hitPiece(boxes: IdBox[], point: Point): string | null {
  for (let index = boxes.length - 1; index >= 0; index -= 1) {
    if (contains(boxes[index].box, point)) return boxes[index].id;
  }
  return null;
}

export function rectFrom(a: Point, b: Point): Box {
  return { x: Math.min(a[0], b[0]), y: Math.min(a[1], b[1]), width: Math.abs(a[0] - b[0]), height: Math.abs(a[1] - b[1]) };
}

export function marqueeHits(boxes: IdBox[], rect: Box): string[] {
  return boxes.filter((item) => overlaps(item.box, rect)).map((item) => item.id);
}

/** The handle within `radius` screen px of the pointer, if any. */
export function handleAt(point: Point, handles: Readonly<Record<Handle, readonly [number, number]>>, radius = 6): Handle | null {
  for (const name of ["nw", "ne", "se", "sw"] as const) {
    const [hx, hy] = handles[name];
    if (Math.abs(point[0] - hx) <= radius && Math.abs(point[1] - hy) <= radius) return name;
  }
  return null;
}

/** Direction from center to point in degrees, clockwise on screen, 0 = toward +x. */
export function angleAt(center: Point, point: Point): number {
  return normalizeAngle((Math.atan2(point[1] - center[1], point[0] - center[0]) * 180) / Math.PI);
}

export function snapAngle(degrees: number, step: number): number {
  return normalizeAngle(Math.round(degrees / step) * step);
}

/** SVG path text for flat lines [x0, y0, x1, y1, …]. */
export function linesToPath(lines: number[][]): string {
  let text = "";
  for (const line of lines) {
    for (let index = 0; index + 1 < line.length; index += 2) {
      text += `${index === 0 ? "M" : "L"}${line[index]} ${line[index + 1]}`;
    }
  }
  return text;
}

/** Multiples of spacing from `from` to `to`. */
export function gridLines(from: number, to: number, spacing: number): number[] {
  const lines: number[] = [];
  for (let value = Math.ceil(from / spacing) * spacing; value <= to; value += spacing) {
    lines.push(Math.round(value / spacing) * spacing + 0); // + 0 turns -0 into 0
  }
  return lines;
}

export function unionBox(boxes: Box[]): Box | null {
  if (boxes.length === 0) return null;
  const x0 = Math.min(...boxes.map((b) => b.x));
  const y0 = Math.min(...boxes.map((b) => b.y));
  const x1 = Math.max(...boxes.map((b) => b.x + b.width));
  const y1 = Math.max(...boxes.map((b) => b.y + b.height));
  return { x: x0, y: y0, width: x1 - x0, height: y1 - y0 };
}

/** Talking to Python through pywebview (docs/bridge.md). */
import type { Api } from "./contract";

/** A problem Python reported: `message` is for the student as written, `details` for the Details link. */
export class BridgeError extends Error {
  readonly details: string;

  constructor(message: string, details = "") {
    super(message);
    this.name = "BridgeError";
    this.details = details;
  }
}

export type Method = keyof Api;
export type Args<M extends Method> = Parameters<Api[M]>;
export type Ok<M extends Method> = Extract<Awaited<ReturnType<Api[M]>>, { ok: true }>;

type AnyCall = (method: Method, ...args: unknown[]) => Promise<unknown>;
type Slot = { method: Method; args: unknown[]; resolve: (value: unknown) => void; reject: (error: unknown) => void };

/**
 * call(): one bridge call; {ok: false} becomes a BridgeError.
 * hardware(): plotter requests run one at a time (Python answers a second one with "still busy").
 * latest(): like hardware(), but while one runs only the newest waiting call for the same key is kept;
 * the calls it replaces resolve to null. The pen-height slider uses it.
 */
export class Client {
  private chain: Promise<unknown> = Promise.resolve();
  private readonly slots = new Map<string, Slot>();
  private readonly draining = new Set<string>();

  constructor(private readonly api: Api) {}

  async call<M extends Method>(method: M, ...args: Args<M>): Promise<Ok<M>> {
    const fn = this.api[method] as unknown as (...a: unknown[]) => Promise<{ ok: boolean; message?: string; details?: string }>;
    const reply = await fn.apply(this.api, args);
    if (!reply.ok) throw new BridgeError(reply.message ?? "", reply.details ?? "");
    return reply as Ok<M>;
  }

  hardware<M extends Method>(method: M, ...args: Args<M>): Promise<Ok<M>> {
    const run = this.chain.then(() => this.call(method, ...args));
    this.chain = run.catch(() => undefined);
    return run;
  }

  latest<M extends Method>(key: string, method: M, ...args: Args<M>): Promise<Ok<M> | null> {
    return new Promise((resolve, reject) => {
      this.slots.get(key)?.resolve(null);
      this.slots.set(key, { method, args, resolve: resolve as (value: unknown) => void, reject });
      if (!this.draining.has(key)) void this.drain(key);
    });
  }

  private async drain(key: string): Promise<void> {
    this.draining.add(key);
    const hardware = this.hardware.bind(this) as unknown as AnyCall;
    try {
      for (let slot = this.slots.get(key); slot; slot = this.slots.get(key)) {
        this.slots.delete(key);
        try {
          slot.resolve(await hardware(slot.method, ...slot.args));
        } catch (error) {
          slot.reject(error);
        }
      }
    } finally {
      this.draining.delete(key);
    }
  }
}

type PyWindow = Window & {
  pywebview?: { api?: Api };
  rosettaEvent?: (name: string, data: unknown) => void;
};

/** The real bridge, once pywebview has injected it. Rejects after timeoutMs (in a plain browser, say). */
export function waitForApi(timeoutMs: number): Promise<Api> {
  const w = window as PyWindow;
  return new Promise((resolve, reject) => {
    if (w.pywebview?.api) {
      resolve(w.pywebview.api);
      return;
    }
    const timer = window.setTimeout(() => reject(new Error("The app's Python side didn't answer.")), timeoutMs);
    window.addEventListener(
      "pywebviewready",
      () => {
        window.clearTimeout(timer);
        if (w.pywebview?.api) resolve(w.pywebview.api);
        else reject(new Error("The app's Python side didn't answer."));
      },
      { once: true },
    );
  });
}

/** Python calls window.rosettaEvent(name, data): plot states, progress, dropped files, closing mid-plot. */
export function listenForEvents(handler: (name: string, data: unknown) => void): void {
  (window as PyWindow).rosettaEvent = handler;
}

export class Store<T extends object> {
  private readonly listeners = new Set<() => void>();

  constructor(public state: T) {}

  subscribe(listener: () => void): () => void {
    this.listeners.add(listener);
    return () => this.listeners.delete(listener);
  }

  update(change: Partial<T> | ((state: T) => Partial<T>)): void {
    const patch = typeof change === "function" ? change(this.state) : change;
    this.state = { ...this.state, ...patch };
    for (const listener of [...this.listeners]) listener();
  }
}

/** Re-renders a Lit component whenever the store changes. */
export class StoreController<T extends object> implements ReactiveController {
  private unsubscribe: (() => void) | null = null;

  constructor(
    private readonly host: ReactiveControllerHost,
    private readonly store: Store<T>,
  ) {
    host.addController(this);
  }

  hostConnected(): void {
    this.unsubscribe = this.store.subscribe(() => this.host.requestUpdate());
  }

  hostDisconnected(): void {
    this.unsubscribe?.();
    this.unsubscribe = null;
  }
}

import { plotRunning } from "./selectors";
import type { AppState } from "./store";

export type Control = "start" | "setHome" | "move" | "trace" | "arrange" | "markCorner" | "rotatePaper" | "model";

const NO_PLOTTER = "No plotter connected. Plug in the USB cable and the power adapter.";
const NO_MODEL = "Model unknown. Choose it in Plotter › Plotter model…";
const NO_POWER = "The power adapter isn't plugged in.";
const NO_HOME = "Home not set. Use Set Home first.";
const NOTHING = "Nothing placed on the paper.";
const NO_PENS = "No pens are turned on in the Pens panel.";
const RUNNING = "A plot is running.";
const NO_SELECTION = "Select a drawing first.";

export function disabledReason(control: Control, s: AppState): string | null {
  const machine = s.machine.state;
  const checks: Record<Control, [boolean, string][]> = {
    start: [
      [machine === "no_plotter", NO_PLOTTER],
      [machine === "needs_model", NO_MODEL],
      [!s.machine.power_ok, NO_POWER],
      [machine === "not_homed", NO_HOME],
      [s.pieces.length === 0, NOTHING],
      [s.colors.length > 0 && s.colors.every((c) => !c.visible), NO_PENS],
      [plotRunning(s), RUNNING],
    ],
    setHome: [
      [machine === "no_plotter", NO_PLOTTER],
      [machine === "needs_model", NO_MODEL],
      [plotRunning(s), RUNNING],
    ],
    move: [
      [machine === "no_plotter", NO_PLOTTER],
      [machine === "needs_model", NO_MODEL],
      [plotRunning(s), RUNNING],
      [machine === "not_homed", NO_HOME],
    ],
    trace: [
      [machine === "no_plotter", NO_PLOTTER],
      [plotRunning(s), RUNNING],
      [machine !== "ready", NO_HOME],
      [s.pieces.length === 0, NOTHING],
    ],
    arrange: [
      [s.selection.length === 0, NO_SELECTION],
      [plotRunning(s), RUNNING],
    ],
    markCorner: [
      [plotRunning(s), RUNNING],
      [machine !== "ready", NO_HOME],
    ],
    rotatePaper: [[plotRunning(s), RUNNING]],
    model: [
      [machine === "no_plotter", NO_PLOTTER],
      [plotRunning(s), RUNNING],
    ],
  };
  return checks[control].find(([applies]) => applies)?.[1] ?? null;

const NO_ROOM = "The margins leave no room on this paper.";
const INFO_MS = 8000;
const DROP_MEMORY_MS = 3000;

export type ViewRequest = { kind: "fit-paper" } | { kind: "fit-reach" } | { kind: "zoom"; factor: number };

export interface AppOptions {
  colorsDelayMs: number;
  scanIntervalMs: number;
}

const DEFAULTS: AppOptions = { colorsDelayMs: 300, scanIntervalMs: 2000 };

function fileName(path: string): string {
  return path.split(/[\\/]/).pop() ?? path;
}

export class App {
  /** The pointer's machine position; kept apart so moving the mouse only re-renders the status bar. */
  readonly cursor = new Store<{ point: Point | null }>({ point: null });
  /** Requests for the plate that aren't state: fit and zoom. */
  readonly viewRequests = new EventTarget();
  /** Where files from Windows were dropped on the plate, so the files_dropped event can place them there. */
  lastFileDrop: { point: Point; at: number } | null = null;

  private readonly options: AppOptions;
  private scanTimer: ReturnType<typeof setInterval> | null = null;
  private colorsTimer: ReturnType<typeof setTimeout> | null = null;
  private noticeTimer: ReturnType<typeof setTimeout> | null = null;

  constructor(
    readonly store: Store<AppState>,
    readonly client: Client,
    options: Partial<AppOptions> = {},
  ) {
    this.options = { ...DEFAULTS, ...options };
  }

  get state(): AppState {
    return this.store.state;
  }

  // Start-up, events and problems.

  async start(): Promise<void> {
    const start = await this.client.call("app_start");
    const { jobs, pieces } = await this.client.call("job_list");
    this.store.update({
      ready: true,
      start,
      units: start.settings.units,
      machine: start.machine,
      material: start.material,
      paper: start.settings.paper,
      margin: start.settings.margin_mm,
      plot: start.plot,
      jobs,
      pieces,
    });
    this.refreshColors();
    this.scanTimer = setInterval(() => {
      if (this.state.machine.state === "no_plotter") void this.scan();
    }, this.options.scanIntervalMs);
    if (start.machine.state === "no_plotter") await this.scan();
    this.requestView({ kind: "fit-paper" });
  }

  stop(): void {
    if (this.scanTimer) clearInterval(this.scanTimer);
    if (this.colorsTimer) clearTimeout(this.colorsTimer);
    if (this.noticeTimer) clearTimeout(this.noticeTimer);
    this.scanTimer = this.colorsTimer = this.noticeTimer = null;
  }

  onEvent(name: string, data: unknown): void {
    if (name === "plot_state") {
      this.store.update({ plot: data as PlotStatus });
    } else if (name === "files_dropped") {
      const recent = this.lastFileDrop && Date.now() - this.lastFileDrop.at < DROP_MEMORY_MS;
      const point = recent && this.lastFileDrop ? this.lastFileDrop.point : undefined;
      this.lastFileDrop = null;
      void this.importFiles((data as { paths: string[] }).paths, point);
    }
  }

  notify(notice: Notice): void {
    if (this.noticeTimer) clearTimeout(this.noticeTimer);
    this.noticeTimer = null;
    this.store.update({ notice });
    if (notice.kind === "info") this.noticeTimer = setTimeout(() => this.dismissNotice(), INFO_MS);
  }

  dismissNotice(): void {
    this.store.update({ notice: null });
  }

  fail(error: unknown): void {
    if (error instanceof BridgeError) {
      this.notify({ kind: "error", message: error.message, details: error.details });
    } else {
      const details = error instanceof Error ? (error.stack ?? error.message) : String(error);
      this.notify({ kind: "error", message: "Something unexpected went wrong in the window.", details });
    }
  }

  /** Say why a greyed-out control can't work yet. */
  explain(reason: string): void {
    this.notify({ kind: "info", message: reason });
  }

  private blocked(control: Control): boolean {
    const reason = disabledReason(control, this.state);
    if (reason) this.explain(reason);
    return reason !== null;
  }

  private async attempt<T>(work: () => Promise<T>): Promise<T | undefined> {
    try {
      return await work();
    } catch (error) {
      this.fail(error);
      return undefined;
    }
  }

  /** A plotter request; afterwards the plotter's state is refreshed either way (bridge.md). */
  private async plotterAction(work: () => Promise<{ machine: MachineStatus }>): Promise<boolean> {
    try {
      this.setMachine((await work()).machine);
      return true;
    } catch (error) {
      this.fail(error);
      await this.scan();
      return false;
    }
  }

  // The plotter.

  setMachine(machine: MachineStatus): void {
    this.store.update((s) => ({ machine, homing: s.homing && machine.state === "not_homed" }));
  }

  /** Look for the plotter. No plotter is normal, so this never shows a problem. */
  async scan(): Promise<void> {
    try {
      this.setMachine((await this.client.call("plotter_scan")).machine);
    } catch {
      // The next scan tries again.
    }
  }

  async identify(model: number, penlift: number, label: string): Promise<void> {
    if (!(await this.plotterAction(() => this.client.hardware("plotter_identify", model, penlift, label)))) return;
    this.store.update({ dialog: null });
    const machine = this.state.machine;
    this.notify({
      kind: "info",
      message: machine.old_firmware
        ? "Saved on this laptop. This plotter's firmware is old, so other laptops will ask for its model too."
        : `Saved. This plotter is now named "${machine.nickname ?? label}".`,
    });
  }

  /** Set Home: the first click lets go of the motors; the second confirms the pen is in the HOME corner. */
  async setHome(): Promise<void> {
    if (this.blocked("setHome")) return;
    if (!this.state.homing) {
      if (await this.plotterAction(() => this.client.hardware("plotter_motors_off"))) this.store.update({ homing: true });
      return;
    }
    this.store.update({ homing: false });
    await this.plotterAction(() => this.client.hardware("plotter_set_home"));
  }

  cancelHoming(): void {
    this.store.update({ homing: false });
  }

  async jog(dx: number, dy: number): Promise<void> {
    if (this.blocked("move")) return;
    const step = this.state.jogStep;
    await this.plotterAction(() => this.client.hardware("plotter_jog", dx * step, dy * step));
  }

  async movePenTo([x, y]: Point): Promise<void> {
    if (this.blocked("move")) return;
    await this.plotterAction(() => this.client.hardware("plotter_goto", x, y));
  }

  movePenHome(): Promise<void> {
    return this.movePenTo([0, 0]);
  }

  async pen(down: boolean): Promise<void> {
    if (this.blocked("move")) return;
    await this.plotterAction(() => this.client.hardware("plotter_pen", down));
  }

  async testMark(): Promise<void> {
    if (this.blocked("move")) return;
    await this.plotterAction(() => this.client.hardware("plotter_test_mark"));
  }

  /** The pen-height slider and pen type: only the newest value is sent while one is on its way. */
  async setPen(penDown: number, penType: string): Promise<void> {
    this.store.update((s) => ({ machine: { ...s.machine, pen_down: penDown, pen_type: penType } }));
    try {
      const reply = await this.client.latest("pen", "pen_set", penDown, penType);
      if (reply) this.setMachine(reply.machine);
    } catch (error) {
      this.fail(error);
      await this.scan();
    }
  }

  setJogStep(step: 1 | 10): void {
    this.store.update({ jogStep: step });
  }

  openModelDialog(): void {
    if (this.blocked("model")) return;
    this.store.update({ dialog: { kind: "model" } });
  }

  showDialog(kind: "shortcuts" | "about"): void {
    this.store.update({ dialog: { kind } });
  }

  closeDialog(): void {
    this.store.update({ dialog: null });
  }

  async copyDiagnostics(): Promise<void> {
    const reply = await this.attempt(() => this.client.call("diagnostics"));
    if (!reply) return;
    try {
      await navigator.clipboard.writeText(reply.text);
      this.notify({ kind: "info", message: "Diagnostics copied. Paste them into a message to whoever is helping you." });
    } catch {
      this.store.update({ dialog: { kind: "text", title: "Diagnostics", text: reply.text } });
    }
  }

  async exit(): Promise<void> {
    await this.attempt(() => this.client.call("quit_confirmed"));
  }

  // Units, tools and the view.

  async setUnits(units: Units): Promise<void> {
    this.store.update({ units });
    await this.attempt(() => this.client.call("set_units", units));
  }

  setTool(tool: Tool): void {
    this.store.update({ tool });
  }

  toggleView(key: keyof ViewOptions): void {
    this.store.update((s) => ({ view: { ...s.view, [key]: !s.view[key] } }));
  }

  toggleLink(): void {
    this.store.update((s) => ({ linkScale: !s.linkScale }));
  }

  requestView(request: ViewRequest): void {
    this.viewRequests.dispatchEvent(new CustomEvent<ViewRequest>("view", { detail: request }));
  }

  // The paper.

  async setPaper(
    size: string,
    orientation: "landscape" | "portrait",
    customWidth: number | null = null,
    customHeight: number | null = null,
  ): Promise<void> {
    const { x, y } = this.state.material;
    const reply = await this.attempt(() => this.client.call("material_set", size, orientation, customWidth, customHeight, x, y));
    if (!reply) return;
    const custom = size === "custom";
    this.store.update({
      material: reply.material,
      paper: {
        size,
        orientation,
        custom_width_mm: custom ? reply.material.width : null,
        custom_height_mm: custom ? reply.material.height : null,
      },
    });
  }

  /** The orientation buttons swap the paper's sides; the drawings stay where they are. */
  async setOrientation(orientation: "landscape" | "portrait"): Promise<void> {
    const { paper, material } = this.state;
    if (paper.orientation === orientation) return;
    if (paper.size === "custom") await this.setPaper("custom", orientation, material.height, material.width);
    else await this.setPaper(paper.size, orientation);
  }

  async movePaper(x: number, y: number): Promise<void> {
    const p = this.state.paper;
    const reply = await this.attempt(() =>
      this.client.call("material_set", p.size, p.orientation, p.custom_width_mm, p.custom_height_mm, x, y),
    );
    if (reply) this.store.update({ material: reply.material });
  }

  async markCorner(): Promise<void> {
    if (this.blocked("markCorner")) return;
    const reply = await this.attempt(() => this.client.call("material_mark_corner"));
    if (reply) this.store.update({ material: reply.material });
  }

  async setMargin(margin: number): Promise<void> {
    const reply = await this.attempt(() => this.client.call("set_margin", margin));
    if (reply) this.store.update({ margin: reply.margin_mm });
  }

  async rotatePaper(): Promise<void> {
    if (this.blocked("rotatePaper")) return;
    const reply = await this.attempt(() => this.client.call("paper_rotate"));
    if (!reply) return;
    this.store.update({ material: reply.material, paper: reply.paper, pieces: reply.pieces });
    this.refreshColors();
  }

  // Jobs.

  /** Import files into the Jobs list: from Windows' Open dialog, or the given paths. Placed at `dropAt` if given. */
  async importFiles(paths?: string[], dropAt?: Point): Promise<void> {
    let files = paths;
    if (!files) {
      const chosen = await this.attempt(() => this.client.call("artwork_choose"));
      if (!chosen) return;
      files = chosen.paths;
    }
    for (const path of files) {
      const counted = await this.attempt(() => this.client.call("artwork_pages", path));
      if (!counted) continue;
      let page = 0;
      if (counted.pages > 1) {
        const chosen = await this.askPage(fileName(path), counted.pages);
        if (chosen === null) continue;
        page = chosen;
      }
      const reply = await this.attempt(() => this.client.call("job_import", path, page));
      if (!reply) continue;
      this.store.update((s) => ({ jobs: [...s.jobs, reply.job] }));
      if (dropAt) await this.placeJob(reply.job.id, dropAt);
    }
  }

  /** Which page of a many-page file to import (0-based), or null if cancelled. */
  askPage(name: string, pages: number): Promise<number | null> {
    return new Promise((resolve) => {
      this.store.update({
        dialog: {
          kind: "pages",
          name,
          pages,
          resolve: (page) => {
            this.store.update({ dialog: null });
            resolve(page);
          },
        },
      });
    });
  }

  /** Where double-clicking a job puts its top-left: the paper's corner nearest Home, inset by the margin. */
  marginCorner(): Point {
    return [this.state.margin, this.state.margin];
  }

  async placeJob(jobId: string, at: Point = this.marginCorner()): Promise<void> {
    if (plotRunning(this.state)) return this.explain(RUNNING);
    const reply = await this.attempt(() => this.client.call("job_place", jobId, at[0], at[1]));
    if (!reply) return;
    this.store.update((s) => ({ pieces: [...s.pieces, reply.piece], selection: [reply.piece.id] }));
    this.refreshColors();
  }

  /** Remove a job; with copies on the paper, ask first in the strip (spec §4.2). */
  removeJob(jobId: string): void {
    const job = this.state.jobs.find((j) => j.id === jobId);
    if (!job) return;
    if (plotRunning(this.state)) return this.explain(RUNNING);
    const copies = this.state.pieces.filter((p) => p.job_id === jobId).length;
    const remove = async () => {
      this.dismissNotice();
      const reply = await this.attempt(() => this.client.call("job_remove", jobId));
      if (!reply) return;
      const gone = new Set(reply.removed_pieces);
      this.store.update((s) => ({
        jobs: s.jobs.filter((j) => j.id !== jobId),
        pieces: s.pieces.filter((p) => !gone.has(p.id)),
        selection: s.selection.filter((id) => !gone.has(id)),
      }));
      this.refreshColors();
    };
    if (copies === 0) {
      void remove();
      return;
    }
    this.notify({
      kind: "warning",
      message: `Remove ${job.name} and its ${copies} ${copies === 1 ? "copy" : "copies"} on the paper?`,
      actions: [
        { label: "Remove", run: () => void remove() },
        { label: "Keep", run: () => this.dismissNotice() },
      ],
    });
  }

  /** The job behind the selected drawing (for Object › Remove job from list). */
  selectedJobId(): string | null {
    return selectedPieces(this.state)[0]?.job_id ?? null;
  }

  // Selection.

  select(ids: string[]): void {
    this.store.update({ selection: ids });
  }

  toggleSelect(id: string): void {
    this.store.update((s) => ({
      selection: s.selection.includes(id) ? s.selection.filter((x) => x !== id) : [...s.selection, id],
    }));
  }

  selectAll(): void {
    this.store.update((s) => ({ selection: s.pieces.map((p) => p.id) }));
  }

  deselect(): void {
    this.store.update({ selection: [] });
  }

  // Placing and editing drawings.

  /** Show a placement while dragging, without telling Python yet. */
  previewPlacement(id: string, placement: PlacementData): void {
    this.store.update((s) => ({ pieces: s.pieces.map((p) => (p.id === id ? { ...p, placement } : p)) }));
  }

  /** Send a piece's current placement to Python (the end of a drag, or a typed value). */
  async commitPlacement(id: string): Promise<void> {
    const piece = this.state.pieces.find((p) => p.id === id);
    if (!piece) return;
    const p = piece.placement;
    const reply = await this.attempt(() => this.client.call("piece_place", id, p.x, p.y, p.scale, p.rotation, p.scale_y));
    if (reply) this.previewPlacement(id, reply.placement);
    this.refreshColors();
  }

  async setPlacement(id: string, placement: PlacementData): Promise<void> {
    this.previewPlacement(id, placement);
    await this.commitPlacement(id);
  }

  async moveSelectionBy(dx: number, dy: number): Promise<void> {
    if (this.blocked("arrange")) return;
    for (const piece of selectedPieces(this.state)) {
      await this.setPlacement(piece.id, { ...piece.placement, x: piece.placement.x + dx, y: piece.placement.y + dy });
    }
  }

  /** Typed X or Y for the selection's box (null leaves that one). */
  async moveSelectionTo(x: number | null, y: number | null): Promise<void> {
    const box = selectionBox(this.state);
    if (!box) {
      this.blocked("arrange");
      return;
    }
    await this.moveSelectionBy(x === null ? 0 : x - box.x, y === null ? 0 : y - box.y);
  }

  private single() {
    if (this.blocked("arrange")) return null;
    const pieces = selectedPieces(this.state);
    if (pieces.length !== 1) {
      this.explain(ONE_DRAWING);
      return null;
    }
    const job = jobOf(this.state, pieces[0]);
    return job ? { piece: pieces[0], job } : null;
  }

  async resizeSelected(size: { width?: number; height?: number }): Promise<void> {
    const one = this.single();
    if (!one) return;
    const next = resizeBox(one.job.width_mm, one.job.height_mm, one.piece.placement, size, this.state.linkScale);
    if (!next) return this.explain(STRETCH_TURNED);
    await this.setPlacement(one.piece.id, next);
  }

  /** W% / H% (or one Scale % while linked); the top-left stays put. */
  async setScale(scale: number, scaleY: number): Promise<void> {
    const one = this.single();
    if (!one) return;
    await this.setPlacement(one.piece.id, { ...one.piece.placement, scale, scale_y: scaleY });
  }

  async rotateSelectedTo(angle: number): Promise<void> {
    const one = this.single();
    if (!one) return;
    await this.setPlacement(one.piece.id, rotateAboutCenter(one.job.width_mm, one.job.height_mm, one.piece.placement, angle));
  }

  /** Rotate 90°: each selected drawing turns about its own center. */
  async rotateSelectionBy(delta: number): Promise<void> {
    if (this.blocked("arrange")) return;
    for (const piece of selectedPieces(this.state)) {
      const job = jobOf(this.state, piece);
      if (!job) continue;
      await this.setPlacement(piece.id, rotateAboutCenter(job.width_mm, job.height_mm, piece.placement, piece.placement.rotation + delta));
    }
  }

  async centerSelection(): Promise<void> {
    const box = selectionBox(this.state);
    if (!box) {
      this.blocked("arrange");
      return;
    }
    const { width, height } = this.state.material;
    await this.moveSelectionBy((width - box.width) / 2 - box.x, (height - box.height) / 2 - box.y);
  }

  /** Fit to paper: shrink (never enlarge) to fit inside the margins, then center (spec §4.2). */
  async fitSelection(): Promise<void> {
    const one = this.single();
    if (!one) return;
    const { width, height } = this.state.material;
    const margin = this.state.margin;
    const box = pieceBoxOnPaper(this.state, one.piece);
    const factor = Math.min(1, (width - 2 * margin) / box.width, (height - 2 * margin) / box.height);
    if (!(factor > 0)) return this.explain(NO_ROOM);
    const p = one.piece.placement;
    const scaled = { ...p, scale: p.scale * factor, scale_y: p.scale_y * factor };
    const size = pieceBox(one.job.width_mm, one.job.height_mm, scaled);
    await this.setPlacement(one.piece.id, { ...scaled, x: (width - size.width) / 2, y: (height - size.height) / 2 });
  }

  async duplicateSelection(): Promise<void> {
    if (this.blocked("arrange")) return;
    const copies: string[] = [];
    for (const piece of selectedPieces(this.state)) {
      const reply = await this.attempt(() => this.client.call("piece_duplicate", piece.id));
      if (!reply) continue;
      copies.push(reply.piece.id);
      this.store.update((s) => {
        const pieces = [...s.pieces];
        pieces.splice(pieces.findIndex((p) => p.id === piece.id) + 1, 0, reply.piece);
        return { pieces };
      });
    }
    if (copies.length > 0) this.select(copies);
    this.refreshColors();
  }

  async deleteSelection(): Promise<void> {
    if (this.blocked("arrange")) return;
    for (const id of [...this.state.selection]) {
      const reply = await this.attempt(() => this.client.call("piece_delete", id));
      if (!reply) continue;
      this.store.update((s) => ({ pieces: s.pieces.filter((p) => p.id !== id), selection: s.selection.filter((x) => x !== id) }));
    }
    this.refreshColors();
  }

  // Pens.

  /** Ask Python for the pen colors, a moment after the last change (spec §4.3). */
  refreshColors(): void {
    if (this.colorsTimer) clearTimeout(this.colorsTimer);
    this.colorsTimer = setTimeout(() => {
      this.colorsTimer = null;
      void this.attempt(async () => {
        const reply = await this.client.call("job_colors");
        this.store.update({ colors: reply.colors });
      });
    }, this.options.colorsDelayMs);
  }

  async setPenVisible(color: string, visible: boolean): Promise<void> {
    this.store.update((s) => ({ colors: s.colors.map((c) => (c.color === color ? { ...c, visible } : c)) }));
    await this.attempt(() => this.client.call("set_pen_visible", color, visible));
    this.refreshColors();
  }
}

/**
 * A number box: a length in the chosen units, a percentage, or an angle. Enter or leaving the box
 * commits with a "commit" event whose detail is the value (mm, a factor, or degrees). Esc, or text it
 * can't read, puts the old value back.
 */
import { css, html, LitElement } from "lit";
import { formatAngle, formatLength, formatPercent, parseAngle, parseLength, parsePercent, type Units } from "../geometry/units";
import { controls } from "./styles";

export type FieldKind = "length" | "percent" | "angle";

export class RpField extends LitElement {
  static properties = {
    value: { type: Number },
    kind: {},
    units: {},
    label: {},
    tip: {},
    disabled: { type: Boolean },
    readonly: { type: Boolean },
    invalid: { state: true },
  };

  static styles = [
    controls,
    css`
      :host {
        display: inline-flex;
        align-items: center;
        gap: 4px;
      }
      label {
        color: var(--text-dim);
      }
      input {
        width: 64px;
        text-align: right;
      }
    `,
  ];

  declare value: number;
  declare kind: FieldKind;
  declare units: Units;
  declare label: string;
  declare tip: string;
  declare disabled: boolean;
  declare readonly: boolean;
  declare invalid: boolean;

  constructor() {
    super();
    this.value = 0;
    this.kind = "length";
    this.units = "in";
    this.label = "";
    this.tip = "";
    this.disabled = false;
    this.readonly = false;
    this.invalid = false;
  }

  private format(): string {
    if (this.kind === "percent") return formatPercent(this.value);
    if (this.kind === "angle") return formatAngle(this.value);
    return formatLength(this.value, this.units);
  }

  private parse(text: string): number | null {
    if (this.kind === "percent") return parsePercent(text);
    if (this.kind === "angle") return parseAngle(text);
    return parseLength(text, this.units);
  }

  private commit(input: HTMLInputElement): void {
    const value = this.parse(input.value);
    if (value === null) {
      this.invalid = true;
      input.value = this.format();
      return;
    }
    this.invalid = false;
    if (Math.abs(value - this.value) > 1e-9) this.dispatchEvent(new CustomEvent("commit", { detail: value }));
    input.value = this.format();
  }

  private onKey(event: KeyboardEvent): void {
    const input = event.target as HTMLInputElement;
    if (event.key === "Enter") {
      this.commit(input);
      input.dataset.committed = "1";
      input.blur();
    } else if (event.key === "Escape") {
      input.value = this.format();
      this.invalid = false;
      input.dataset.committed = "1";
      input.blur();
    }
  }

  private onBlur(event: FocusEvent): void {
    const input = event.target as HTMLInputElement;
    if (input.dataset.committed) {
      delete input.dataset.committed;
      return;
    }
    this.commit(input);
  }

  render() {
    return html`${this.label ? html`<label for="f">${this.label}</label>` : ""}<input
        id="f"
        .value=${this.format()}
        ?disabled=${this.disabled}
        ?readonly=${this.readonly}
        title=${this.tip}
        aria-label=${this.label || this.tip}
        aria-invalid=${this.invalid ? "true" : "false"}
        @keydown=${this.onKey}
        @focus=${(e: FocusEvent) => delete (e.target as HTMLInputElement).dataset.committed}
        @blur=${this.onBlur}
        @input=${() => (this.invalid = false)}
      />`;
  }
}

export class RpPlate extends AppElement {

  private camera: Camera = { scale: 1, x: 0, y: 0 };
  private size: Size = { width: 800, height: 600 };
  private drag: Drag | null = null;
  private spaceHeld = false;
  private fitted = false;
  private observer: ResizeObserver | null = null;
  private readonly paths = new Map<string, string>();

  /** The current camera (for tests and the status bar's future use). */
  get view(): Camera {
    return this.camera;
  }

  connectedCallback(): void {
    super.connectedCallback();
    this.tabIndex = 0;
    this.app.viewRequests.addEventListener("view", this.onViewRequest);
    window.addEventListener("keydown", this.onSpace);
    window.addEventListener("keyup", this.onSpace);
    if (typeof ResizeObserver !== "undefined") {
      this.observer = new ResizeObserver(() => this.measure());
      this.observer.observe(this);
    }
  }

  disconnectedCallback(): void {
    this.app.viewRequests.removeEventListener("view", this.onViewRequest);
    window.removeEventListener("keydown", this.onSpace);
    window.removeEventListener("keyup", this.onSpace);
    this.observer?.disconnect();
    this.observer = null;
    super.disconnectedCallback();
  }

  protected firstUpdated(): void {
    this.measure();
  }

  protected updated(): void {
    this.dataset.tool = this.activeTool();
  }

  private measure(): void {
    const rect = this.getBoundingClientRect();
    if (rect.width <= 0 || rect.height <= 0) return;
    this.size = { width: rect.width, height: rect.height };
    if (this.fitted) this.requestUpdate();
    else this.fit("paper");
  }

  private fit(what: "paper" | "reach"): void {
    const { material } = this.s;
    const reach = reachOf(this.s);
    const box: Box =
      what === "reach" && reach
        ? { x: 0, y: 0, width: reach[0], height: reach[1] }
        : { x: material.x, y: material.y, width: material.width, height: material.height };
    this.camera = fitCamera(box, this.size);
    this.fitted = true;
    this.requestUpdate();
  }

  private onViewRequest = (event: Event): void => {
    const request = (event as CustomEvent<ViewRequest>).detail;
    if (request.kind === "zoom") {
      this.camera = zoomAt(this.camera, request.factor, [this.size.width / 2, this.size.height / 2]);
      this.requestUpdate();
    } else {
      this.fit(request.kind === "fit-reach" ? "reach" : "paper");
    }
  };

  private onSpace = (event: KeyboardEvent): void => {
    if (event.code !== "Space" || isTyping(event)) return;
    event.preventDefault();
    const held = event.type === "keydown";
    if (held !== this.spaceHeld) {
      this.spaceHeld = held;
      this.requestUpdate();
    }
  };

  private activeTool(): Tool {
    return this.spaceHeld ? "hand" : this.s.tool;
  }

  private screenPoint(event: MouseEvent): Point {
    const rect = this.getBoundingClientRect();
    return [event.clientX - rect.left, event.clientY - rect.top];
  }

  private paperPoint(screen: Point): Point {
    const [x, y] = toMachine(this.camera, screen);
    return [x - this.s.material.x, y - this.s.material.y];
  }

  private boxes(): IdBox[] {
    return this.s.pieces.map((piece) => ({ id: piece.id, box: pieceBoxOnPaper(this.s, piece) }));
  }

  private selectedOne(): PieceInfo | null {
    return this.s.selection.length === 1 ? (this.s.pieces.find((p) => p.id === this.s.selection[0]) ?? null) : null;
  }

  /** Screen positions of the one selected drawing's corner handles. */
  private handles(): Record<Handle, Point> | null {
    const piece = this.selectedOne();
    if (!piece) return null;
    const box = pieceBoxOnPaper(this.s, piece);
    const { x, y } = this.s.material;
    const corner = (cx: number, cy: number) => toScreen(this.camera, [x + cx, y + cy]);
    return {
      nw: corner(box.x, box.y),
      ne: corner(box.x + box.width, box.y),
      se: corner(box.x + box.width, box.y + box.height),
      sw: corner(box.x, box.y + box.height),
    };
  }

  private capture(event: PointerEvent): void {
    try {
      (event.currentTarget as Element | null)?.setPointerCapture?.(event.pointerId);
    } catch {
      // Not every pointer can be captured; dragging still works without it.
    }
  }

  private onDown = (event: PointerEvent): void => {
    const screen = this.screenPoint(event);
    const tool = this.activeTool();
    if (event.button === 1 || tool === "hand") {
      this.drag = { kind: "pan", last: screen };
      this.capture(event);
      return;
    }
    if (event.button !== 0) return;
    this.focus();
    const s = this.s;
    if (tool === "zoom") {
      this.camera = zoomAt(this.camera, event.altKey ? 1 / 1.5 : 1.5, screen);
      this.requestUpdate();
      return;
    }
    if (tool === "pen") {
      void this.app.movePenTo(toMachine(this.camera, screen));
      return;
    }
    const paper = this.paperPoint(screen);
    const locked = plotRunning(s);
    if (tool === "rotate") {
      const piece = this.selectedOne();
      const reason = disabledReason("arrange", s) ?? (piece ? null : ONE_DRAWING);
      if (reason || !piece) {
        this.app.explain(reason ?? ONE_DRAWING);
        return;
      }
      const box = pieceBoxOnPaper(s, piece);
      const center: Point = [box.x + box.width / 2, box.y + box.height / 2];
      this.drag = { kind: "rotate", id: piece.id, center, startAngle: angleAt(center, paper), original: piece.placement };
      this.capture(event);
      return;
    }
    const handles = locked ? null : this.handles();
    const handle = handles ? handleAt(screen, handles) : null;
    const one = this.selectedOne();
    if (handle && one) {
      this.drag = { kind: "scale", id: one.id, handle, original: one.placement };
      this.capture(event);
      return;
    }
    const hit = hitPiece(this.boxes(), paper);
    if (hit) {
      if (event.shiftKey) {
        this.app.toggleSelect(hit);
        return;
      }
      if (!s.selection.includes(hit)) this.app.select([hit]);
      if (locked) return;
      const selected = this.s.pieces.filter((p) => this.s.selection.includes(p.id));
      this.drag = { kind: "move", start: paper, originals: new Map(selected.map((p) => [p.id, p.placement])) };
      this.capture(event);
      return;
    }
    if (!event.shiftKey) this.app.deselect();
    this.drag = { kind: "marquee", start: screen, end: screen, additive: event.shiftKey };
    this.capture(event);
  };

  private onMove = (event: PointerEvent): void => {
    const screen = this.screenPoint(event);
    this.app.cursor.update({ point: toMachine(this.camera, screen) });
    const drag = this.drag;
    if (!drag) return;
    if (drag.kind === "pan") {
      this.camera = panBy(this.camera, screen[0] - drag.last[0], screen[1] - drag.last[1]);
      drag.last = screen;
      this.requestUpdate();
      return;
    }
    if (drag.kind === "marquee") {
      drag.end = screen;
      this.requestUpdate();
      return;
    }
    const paper = this.paperPoint(screen);
    if (drag.kind === "move") {
      let dx = paper[0] - drag.start[0];
      let dy = paper[1] - drag.start[1];
      if (event.shiftKey) {
        if (Math.abs(dx) >= Math.abs(dy)) dy = 0;
        else dx = 0;
      }
      for (const [id, original] of drag.originals) this.app.previewPlacement(id, { ...original, x: original.x + dx, y: original.y + dy });
      return;
    }
    const piece = this.s.pieces.find((p) => p.id === drag.id);
    const job = piece ? jobOf(this.s, piece) : undefined;
    if (!job) return;
    if (drag.kind === "scale") {
      this.app.previewPlacement(drag.id, scaleFromHandle(job.width_mm, job.height_mm, drag.original, drag.handle, paper, event.shiftKey));
      return;
    }
    const turned = drag.original.rotation + angleAt(drag.center, paper) - drag.startAngle;
    const angle = event.shiftKey ? snapAngle(turned, 15) : turned;
    this.app.previewPlacement(drag.id, rotateAboutCenter(job.width_mm, job.height_mm, drag.original, angle));
  };

  private onUp = (): void => {
    const drag = this.drag;
    this.drag = null;
    if (!drag) return;
    if (drag.kind === "move") {
      for (const [id, original] of drag.originals) {
        const now = this.s.pieces.find((p) => p.id === id)?.placement;
        if (now && (now.x !== original.x || now.y !== original.y)) void this.app.commitPlacement(id);
      }
    } else if (drag.kind === "scale" || drag.kind === "rotate") {
      void this.app.commitPlacement(drag.id);
    } else if (drag.kind === "marquee") {
      if (Math.hypot(drag.end[0] - drag.start[0], drag.end[1] - drag.start[1]) > MARQUEE_MIN_PX) {
        const hits = marqueeHits(this.boxes(), rectFrom(this.paperPoint(drag.start), this.paperPoint(drag.end)));
        this.app.select(drag.additive ? [...new Set([...this.s.selection, ...hits])] : hits);
      }
      this.requestUpdate();
    }
  };

  private onWheel = (event: WheelEvent): void => {
    event.preventDefault();
    this.camera = zoomAt(this.camera, Math.exp(-event.deltaY * WHEEL_ZOOM), this.screenPoint(event));
    this.requestUpdate();
  };

  private onDragOver = (event: DragEvent): void => {
    if (Array.from(event.dataTransfer?.types ?? []).includes(JOB_MIME)) {
      event.preventDefault();
      if (event.dataTransfer) event.dataTransfer.dropEffect = "copy";
    }
  };

  private onDrop = (event: DragEvent): void => {
    const at = this.paperPoint(this.screenPoint(event));
    const jobId = event.dataTransfer?.getData(JOB_MIME);
    if (jobId) {
      event.preventDefault();
      void this.app.placeJob(jobId, at);
    } else if (Array.from(event.dataTransfer?.types ?? []).includes("Files")) {
      // pywebview hands Windows files to Python; its files_dropped event then places them here.
      this.app.lastFileDrop = { point: at, at: Date.now() };
    }
  };

  render() {
    const s = this.s;
    const cam = this.camera;
    const m = s.material;
    const reach = reachOf(s);
    const clipRight = reach ? Math.min(m.x + m.width, reach[0]) : m.x + m.width;
    const clipBottom = reach ? Math.min(m.y + m.height, reach[1]) : m.y + m.height;
    return html`<svg
      role="img"
      aria-label="The plotter's reach, the paper and the drawings on it"
      @pointerdown=${this.onDown}
      @pointermove=${this.onMove}
      @pointerup=${this.onUp}
      @pointercancel=${this.onUp}
      @pointerleave=${() => this.app.cursor.update({ point: null })}
      @wheel=${this.onWheel}
      @dragover=${this.onDragOver}
      @drop=${this.onDrop}
    >
      <defs>
        <clipPath id="printable" clipPathUnits="userSpaceOnUse">
          <rect x=${m.x} y=${m.y} width=${Math.max(0, clipRight - m.x)} height=${Math.max(0, clipBottom - m.y)} />
        </clipPath>
      </defs>
      <g transform=${`matrix(${cam.scale} 0 0 ${cam.scale} ${cam.x} ${cam.y})`}>
        ${s.view.grid ? this.grid() : nothing}
        ${reach ? svg`<rect class="reach" x="0" y="0" width=${reach[0]} height=${reach[1]} />` : nothing}
        <rect class="paper" x=${m.x} y=${m.y} width=${m.width} height=${m.height} />
        ${this.unreachable(reach)}
        ${s.view.margins
          ? svg`<rect class="margin" x=${m.x + s.margin} y=${m.y + s.margin} width=${Math.max(0, m.width - 2 * s.margin)} height=${Math.max(0, m.height - 2 * s.margin)} />`
          : nothing}
        ${this.axes(reach)}
        ${s.pieces.map((piece) => this.piece(piece, "off"))}
        <g clip-path="url(#printable)">${s.pieces.map((piece) => this.piece(piece, "ink"))}</g>
      </g>
      ${this.selection()} ${this.crosshair()} ${this.marquee()}
    </svg>`;
  }

  private grid() {
    const spacing = GRID_MM[this.s.units];
    const [x0, y0] = toMachine(this.camera, [0, 0]);
    const [x1, y1] = toMachine(this.camera, [this.size.width, this.size.height]);
    const lines = (step: number, kind: string) => {
      if (step * this.camera.scale < MIN_GRID_PX) return nothing;
      return svg`${gridLines(x0, x1, step).map((x) => svg`<line class=${kind} x1=${x} y1=${y0} x2=${x} y2=${y1} />`)}${gridLines(y0, y1, step).map(
        (y) => svg`<line class=${kind} x1=${x0} y1=${y} x2=${x1} y2=${y} />`,
      )}`;
    };
    return svg`${lines(spacing, "minor")}${lines(spacing * 5, "major")}`;
  }

  private unreachable(reach: [number, number] | null) {
    if (!reach) return nothing;
    const m = this.s.material;
    const right = m.x + m.width;
    const bottom = m.y + m.height;
    const parts: Box[] = [];
    if (right > reach[0]) parts.push({ x: Math.max(m.x, reach[0]), y: m.y, width: right - Math.max(m.x, reach[0]), height: m.height });
    if (bottom > reach[1]) {
      parts.push({ x: m.x, y: Math.max(m.y, reach[1]), width: Math.min(right, reach[0]) - m.x, height: bottom - Math.max(m.y, reach[1]) });
    }
    return parts
      .filter((b) => b.width > 0 && b.height > 0)
      .map((b) => svg`<rect class="unreachable" x=${b.x} y=${b.y} width=${b.width} height=${b.height} />`);
  }

  private axes(reach: [number, number] | null) {
    const m = this.s.material;
    const xEnd = Math.max(reach?.[0] ?? 0, m.x + m.width);
    const yEnd = Math.max(reach?.[1] ?? 0, m.y + m.height);
    return svg`<line class="axis-x" x1="0" y1="0" x2=${xEnd} y2="0" /><line class="axis-y" x1="0" y1="0" x2="0" y2=${yEnd} />`;
  }

  private path(jobId: string, color: string, lines: number[][]): string {
    const key = `${jobId}|${color}`;
    let d = this.paths.get(key);
    if (d === undefined) {
      d = linesToPath(lines);
      this.paths.set(key, d);
    }
    return d;
  }

  /** One drawing: "off" draws every line red; "ink" draws them again in pen colors, clipped to what can be drawn. */
  private piece(piece: PieceInfo, pass: "off" | "ink") {
    const s = this.s;
    const job = jobOf(s, piece);
    if (!job) return nothing;
    const matrix = pieceMatrix(job.width_mm, job.height_mm, piece.placement, [s.material.x, s.material.y]);
    const hidden = new Set(s.colors.filter((c) => !c.visible).map((c) => c.color));
    const groups = job.groups.filter((g) => !g.white && (pass === "ink" || !hidden.has(g.color)));
    return svg`<g transform=${`matrix(${matrix.join(" ")})`}>
      ${pass === "ink" && s.view.original && job.preview
        ? svg`<image href=${job.preview} x="0" y="0" width=${job.width_mm} height=${job.height_mm} opacity="0.3" preserveAspectRatio="none" />`
        : nothing}
      ${groups.map((group) => {
        const d = this.path(job.id, group.color, group.lines);
        return pass === "off"
          ? svg`<path class="off" d=${d} />`
          : svg`<path class=${hidden.has(group.color) ? "ink hidden" : "ink"} d=${d} style=${`stroke: ${inkColor(group.color)}`} />`;
      })}
    </g>`;
  }

  private selection() {
    const s = this.s;
    const { x, y } = s.material;
    const outlines = s.pieces
      .filter((p) => s.selection.includes(p.id))
      .map((piece) => {
        const box = pieceBoxOnPaper(s, piece);
        const [sx, sy] = toScreen(this.camera, [x + box.x, y + box.y]);
        return svg`<rect class="selection" x=${sx} y=${sy} width=${box.width * this.camera.scale} height=${box.height * this.camera.scale} />`;
      });
    const handles = plotRunning(s) || s.tool === "rotate" ? null : this.handles();
    return svg`${outlines}${handles
      ? Object.values(handles).map(([hx, hy]) => svg`<rect class="handle" x=${hx - HANDLE_PX / 2} y=${hy - HANDLE_PX / 2} width=${HANDLE_PX} height=${HANDLE_PX} />`)
      : nothing}`;
  }

  private crosshair() {
    const position = this.s.machine.position;
    if (!position) return nothing;
    const [cx, cy] = toScreen(this.camera, position);
    const r = CROSS_PX + 6;
    return svg`<g>
      <line class="cross" x1=${cx - r} y1=${cy} x2=${cx - 3} y2=${cy} />
      <line class="cross" x1=${cx + 3} y1=${cy} x2=${cx + r} y2=${cy} />
      <line class="cross" x1=${cx} y1=${cy - r} x2=${cx} y2=${cy - 3} />
      <line class="cross" x1=${cx} y1=${cy + 3} x2=${cx} y2=${cy + r} />
      <rect class="cross-box" x=${cx - 3} y=${cy - 3} width="6" height="6" />
    </g>`;
  }

  private marquee() {
    const drag = this.drag;
    if (drag?.kind !== "marquee") return nothing;
    const box = rectFrom(drag.start, drag.end);
    return svg`<rect class="marquee" x=${box.x} y=${box.y} width=${box.width} height=${box.height} />`;
  }
}

customElements.define("rp-plate", RpPlate);

/** The control bar (spec §2.2): the selected drawing's numbers, or the paper's settings. */
import { css, html, nothing } from "lit";
import { ONE_DRAWING, STRETCH_TURNED } from "../app/app";
import { isQuarterTurn } from "../geometry/placement";
import { disabledReason } from "../state/reasons";
import { jobOf, selectedPieces, selectionBox } from "../state/selectors";
import { AppElement } from "./base";
import "./field";
import { icon } from "./icons";
import { controls } from "./styles";

type Commit = CustomEvent<number>;

export class RpControlbar extends AppElement {
  static styles = [
    controls,
    css`
      :host {
        display: flex;
        align-items: center;
        gap: 10px;
        padding: 0 10px;
        overflow: hidden;
      }
      .name {
        max-width: 180px;
        overflow: hidden;
        text-overflow: ellipsis;
        white-space: nowrap;
        color: var(--text-dim);
      }
      .sep {
        width: 1px;
        align-self: stretch;
        margin: 6px 0;
        background: var(--line);
      }
      label {
        display: inline-flex;
        align-items: center;
        gap: 4px;
        color: var(--text-dim);
      }
      label select {
        color: var(--text);
      }
    `,
  ];

  render() {
    return this.s.selection.length > 0 ? this.drawingControls() : this.paperControls();
  }

  private drawingControls() {
    const s = this.s;
    const units = s.units;
    const box = selectionBox(s);
    if (!box) return nothing;
    const pieces = selectedPieces(s);
    const one = pieces.length === 1 ? pieces[0] : null;
    const job = one ? jobOf(s, one) : undefined;
    const p = one?.placement;
    const locked = disabledReason("arrange", s);
    const onlyOne = one ? null : ONE_DRAWING;
    const turned = p ? !isQuarterTurn(p.rotation) : false;
    const sizeReason = onlyOne ?? locked ?? (!s.linkScale && turned ? STRETCH_TURNED : null);
    const twoScales = p ? !s.linkScale || Math.abs(p.scale - p.scale_y) > 1e-9 : false;
    return html`<span class="name" title=${job?.name ?? ""}>${job ? job.name : `${pieces.length} drawings`}</span>
      <rp-field label="X" .value=${box.x} .units=${units} ?disabled=${!!locked} tip=${locked ?? "Left edge, from the paper's corner"}
        @commit=${(e: Commit) => this.app.moveSelectionTo(e.detail, null)}></rp-field>
      <rp-field label="Y" .value=${box.y} .units=${units} ?disabled=${!!locked} tip=${locked ?? "Top edge, from the paper's corner"}
        @commit=${(e: Commit) => this.app.moveSelectionTo(null, e.detail)}></rp-field>
      <span class="sep"></span>
      <rp-field label="W" .value=${box.width} .units=${units} ?disabled=${!!sizeReason} tip=${sizeReason ?? "Width"}
        @commit=${(e: Commit) => this.app.resizeSelected({ width: e.detail })}></rp-field>
      <button
        class="icon"
        aria-label=${s.linkScale ? "Proportions linked (click to unlink)" : "Proportions unlinked (click to link)"}
        aria-pressed=${String(s.linkScale)}
        @click=${() => this.app.toggleLink()}
      >
        ${icon(s.linkScale ? "link" : "unlink")}
      </button>
      <rp-field label="H" .value=${box.height} .units=${units} ?disabled=${!!sizeReason} tip=${sizeReason ?? "Height"}
        @commit=${(e: Commit) => this.app.resizeSelected({ height: e.detail })}></rp-field>
      <span class="sep"></span>
      ${p
        ? twoScales
          ? html`<rp-field kind="percent" label="W%" .value=${p.scale} ?disabled=${!!locked} tip=${locked ?? "Scale along the drawing's width"}
                @commit=${(e: Commit) => this.app.setScale(e.detail, p.scale_y)}></rp-field>
              <rp-field kind="percent" label="H%" .value=${p.scale_y} ?disabled=${!!locked} tip=${locked ?? "Scale along the drawing's height"}
                @commit=${(e: Commit) => this.app.setScale(p.scale, e.detail)}></rp-field>`
          : html`<rp-field kind="percent" label="Scale" .value=${p.scale} ?disabled=${!!locked} tip=${locked ?? "Scale, keeping proportions"}
              @commit=${(e: Commit) => this.app.setScale(e.detail, e.detail)}></rp-field>`
        : nothing}
      <rp-field kind="angle" label="Angle" .value=${p?.rotation ?? 0} ?disabled=${!!(onlyOne ?? locked)}
        tip=${onlyOne ?? locked ?? "Turn clockwise, in degrees"} @commit=${(e: Commit) => this.app.rotateSelectedTo(e.detail)}></rp-field>`;
  }

  private paperControls() {
    const s = this.s;
    const units = s.units;
    const m = s.material;
    const custom = s.paper.size === "custom";
    const portrait = s.paper.orientation === "portrait";
    const mark = disabledReason("markCorner", s);
    return html`<label
        >Paper
        <select aria-label="Paper size" @change=${(e: Event) => this.choosePaper((e.target as HTMLSelectElement).value)}>
          ${(s.start?.papers ?? []).map((p) => html`<option value=${p.key} ?selected=${p.key === s.paper.size}>${p.label}</option>`)}
          <option value="custom" ?selected=${custom}>Custom</option>
        </select></label
      >
      <button aria-pressed=${String(!portrait)} @click=${() => this.app.setOrientation("landscape")}>Landscape</button>
      <button aria-pressed=${String(portrait)} @click=${() => this.app.setOrientation("portrait")}>Portrait</button>
      <rp-field label="W" .value=${m.width} .units=${units} ?readonly=${!custom} tip=${custom ? "Paper width" : "Choose Custom to type a size"}
        @commit=${(e: Commit) => this.app.setPaper("custom", s.paper.orientation, e.detail, m.height)}></rp-field>
      <rp-field label="H" .value=${m.height} .units=${units} ?readonly=${!custom} tip=${custom ? "Paper height" : "Choose Custom to type a size"}
        @commit=${(e: Commit) => this.app.setPaper("custom", s.paper.orientation, m.width, e.detail)}></rp-field>
      <span class="sep"></span>
      <rp-field label="X" .value=${m.x} .units=${units} tip="The paper's corner nearest HOME, measured from Home"
        @commit=${(e: Commit) => this.app.movePaper(e.detail, m.y)}></rp-field>
      <rp-field label="Y" .value=${m.y} .units=${units} tip="The paper's corner nearest HOME, measured from Home"
        @commit=${(e: Commit) => this.app.movePaper(m.x, e.detail)}></rp-field>
      <button
        aria-disabled=${mark ? "true" : "false"}
        title=${mark ?? "Set the paper's corner nearest HOME to where the pen is"}
        @click=${() => (mark ? this.app.explain(mark) : void this.app.markCorner())}
      >
        ${icon("mark")}Mark corner at pen
      </button>
      <span class="sep"></span>
      <rp-field label="Margin" .value=${s.margin} .units=${units} tip="Guides only; also where double-click places a drawing"
        @commit=${(e: Commit) => this.app.setMargin(e.detail)}></rp-field>`;
  }

  private choosePaper(size: string): void {
    const s = this.s;
    if (size === "custom") void this.app.setPaper("custom", s.paper.orientation, s.material.width, s.material.height);
    else void this.app.setPaper(size, s.paper.orientation);
  }
}

import { css, html, nothing } from "lit";
import type { JobInfo } from "../bridge/contract";
import { JOB_MIME } from "../canvas/interaction";

function fileIcon(path: string): IconName {
  const lower = path.toLowerCase();
  if (lower.endsWith(".pdf") || lower.endsWith(".ai")) return "filePdf";
  return lower.endsWith(".svg") ? "fileVector" : "file";
}

interface Menu {

  x: number;
  y: number;

    super();
    this.menu = null;
  }

    window.addEventListener("pointerdown", this.closeMenu);

    window.removeEventListener("pointerdown", this.closeMenu);
    super.disconnectedCallback();
  }

  private closeMenu = (event: Event): void => {
    const inMenu = event.composedPath().some((node) => node instanceof HTMLElement && node.classList.contains("menu"));
    if (this.menu && !inMenu) this.menu = null;
  };

    const s = this.s;
    const selectedJobs = new Set(selectedPieces(s).map((p) => p.job_id));
    return html`<button @click=${() => this.app.importFiles()}>${icon("plus")}Import</button>
      ${s.jobs.length === 0 ? html`<div class="empty">Import SVG, PDF or AI files, or drop them on the window.</div>` : nothing}
      ${s.jobs.map((job) => this.row(job, selectedJobs.has(job.id)))} ${this.menu ? this.contextMenu(this.menu) : nothing}`;
  }

  private row(job: JobInfo, selected: boolean) {
    const copies = this.s.pieces.filter((p) => p.job_id === job.id).length;
    const issues = job.issues.map((i) => i.message).join("\n");
    return html`<div
      class="row ${selected ? "selected" : ""}"
      role="button"
      tabindex="0"
      draggable="true"
      title=${`${job.name}: double-click, or drag onto the paper, to place it`}
      @dblclick=${() => this.app.placeJob(job.id)}
      @keydown=${(e: KeyboardEvent) => e.key === "Enter" && this.app.placeJob(job.id)}
      @dragstart=${(e: DragEvent) => {
        e.dataTransfer?.setData(JOB_MIME, job.id);
        if (e.dataTransfer) e.dataTransfer.effectAllowed = "copy";
      }}
      @contextmenu=${(e: MouseEvent) => {
        e.preventDefault();
        this.menu = { jobId: job.id, x: e.clientX, y: e.clientY };
      }}
    >
      ${icon(fileIcon(job.path))}<span class="name">${job.name}</span>
      ${copies ? html`<span class="check" title=${`${copies} on the paper`}>${icon("check")}</span>` : nothing}
      ${issues ? html`<span class="warn" title=${issues}>${icon("alert")}</span>` : nothing}
    </div>`;
  }

  private contextMenu(menu: Menu) {
    const job = this.s.jobs.find((j) => j.id === menu.jobId);
    if (!job) return nothing;
    const choose = (run: () => void) => () => {
      this.menu = null;
      run();
    };
    return html`<div class="menu" role="menu" style=${`left: ${menu.x}px; top: ${menu.y}px`}>
      <button role="menuitem" @click=${choose(() => void this.app.placeJob(job.id))}>Place</button>
      <button role="menuitem" @click=${choose(() => this.app.removeJob(job.id))}>Remove from list</button>
      <button
        role="menuitem"
        ?disabled=${job.issues.length === 0}
        @click=${choose(() => this.app.notify({ kind: "warning", message: job.issues.map((i) => i.message).join(" ") }))}
      >Show issues</button>
    </div>`;
  }
}

  constructor() {
    super();
    this.collapsed = new Set();

    </section>`;

    return html`${this.panel("jobs", "Jobs", html`<rp-jobs></rp-jobs>`)}${this.panel("pens", "Pens", html`<rp-pens></rp-pens>`)}${this.panel(
      "plotter",
      "Plotter",
      html`<rp-plotter></rp-plotter>`,

}

const TOOL_KEYS: Record<string, Tool> = { v: "select", h: "hand", z: "zoom", r: "rotate", p: "pen" };
const ARROWS: Record<string, [number, number]> = { arrowleft: [-1, 0], arrowright: [1, 0], arrowup: [0, -1], arrowdown: [0, 1] };

function isTyping(event: KeyboardEvent): boolean {
  const target = event.composedPath()[0];
  return target instanceof HTMLElement && (["INPUT", "SELECT", "TEXTAREA"].includes(target.tagName) || target.isContentEditable);
}

/** Handle one key press. True if it was a shortcut, so the caller stops the browser's own action for it. */
export function handleShortcut(app: App, event: KeyboardEvent): boolean {
  if (event.key === "Escape") {
    if (app.state.homing) app.cancelHoming();
    else if (!app.state.dialog) app.deselect();
    return false;
  }
  if (isTyping(event)) return false;
  const key = event.key.toLowerCase();
  if (event.ctrlKey || event.metaKey) {
    if (key === "i") void app.importFiles();
    else if (key === "d") void app.duplicateSelection();
    else if (key === "r" && event.shiftKey) void app.rotatePaper();
    else if (key === "r") void app.rotateSelectionBy(90);
    else if (key === "c" && event.shiftKey) void app.centerSelection();
    else if (key === "a" && event.shiftKey) app.deselect();
    else if (key === "a") app.selectAll();
    else if (key === "0") app.requestView({ kind: event.altKey ? "fit-reach" : "fit-paper" });
    else if (key === "=" || key === "+") app.requestView({ kind: "zoom", factor: 1.25 });
    else if (key === "-") app.requestView({ kind: "zoom", factor: 0.8 });
    else return false;
    return true;
  }
  if (event.altKey) return false;
  if (key === "delete" || key === "backspace") {
    if (app.state.selection.length > 0) void app.deleteSelection();
    return true;
  }
  if (key in ARROWS) {
    if (app.state.selection.length === 0) return false;
    const step = event.shiftKey ? 10 : 1;
    const [dx, dy] = ARROWS[key];
    void app.moveSelectionBy(dx * step, dy * step);
    return true;
  }
  if (!event.shiftKey && key in TOOL_KEYS) {
    app.setTool(TOOL_KEYS[key]);
    return true;
  }
  return false;
}

export function installShortcuts(app: App): () => void {
  const listener = (event: KeyboardEvent) => {
    if (handleShortcut(app, event)) event.preventDefault();
  };
  window.addEventListener("keydown", listener);
  return () => window.removeEventListener("keydown", listener);
}

/** Start the page: find Python's bridge (or the demo FakeApi under `npm run dev`), then mount the window. */
import "./theme.css";
import "./ui/app-shell";
import { App } from "./app/app";
import { setApp } from "./app/context";
import { Client, listenForEvents, waitForApi } from "./bridge/client";
import type { Api } from "./bridge/contract";
import { FakeApi } from "./bridge/fake";
import { installShortcuts } from "./shortcuts";
import { initialState, Store } from "./state/store";

async function pickApi(): Promise<Api> {
  try {
    return await waitForApi(import.meta.env.DEV ? 1500 : 15000);
  } catch (error) {
    if (import.meta.env.DEV) return new FakeApi(); // a plain browser: run against the demo backend
    throw error;
  }
}

async function boot(): Promise<void> {
  const app = new App(new Store(initialState()), new Client(await pickApi()));
  setApp(app);
  listenForEvents((name, data) => app.onEvent(name, data));
  installShortcuts(app);
  document.body.append(document.createElement("rp-app"));
  await app.start();
  (window as Window & { rosettaReady?: boolean }).rosettaReady = true;
}

boot().catch((error: unknown) => {
  document.body.textContent = `Rosetta Plotter couldn't start: ${error instanceof Error ? error.message : String(error)}`;
});

"""Open the real Rosetta window for a moment, check it works, then close it.

    .venv\\Scripts\\python scripts\\smoke_window.py          the bridge console: Python <-> page both ways
    .venv\\Scripts\\python scripts\\smoke_window.py --page   the built student page (run `npm run build` in ui\\ first)

Not part of pytest: it needs a desktop session and WebView2. The page check uses a stand-in plotter,
so it never touches a real one even if it's plugged in.
"""
import sys
import tempfile
import time
from pathlib import Path

import webview

from rosetta.app.api import Api
from rosetta.app.events import WindowSink
from rosetta.app.machine import Machine
from rosetta.app.main import REPO_ROOT, console_html
from rosetta.app.settings import SettingsStore

READY_TIMEOUT_S = 15.0


class _NoPlotter:
    """Stands in for the plotter: this check must never touch hardware."""

    def connect(self, settings) -> bool:
        return False

    def disconnect(self) -> None:
        pass

    def request_pause(self) -> None:
        pass


def _api() -> tuple[Api, WindowSink]:
    store = SettingsStore(Path(tempfile.mkdtemp()) / "settings.json")
    sink = WindowSink()
    return Api(store, sink, machine=Machine(store, plotter_factory=_NoPlotter)), sink


def check_console() -> bool:
    api, sink = _api()
    window = webview.create_window("Rosetta smoke test", html=console_html(), js_api=api)
    sink.attach(window.evaluate_js)
    seen = {}

    def check():
        try:
            seen["methods"] = window.evaluate_js("Object.keys(window.pywebview.api).length")
            sink.emit("smoke", {"hello": "page"})
            time.sleep(1)
            seen["events"] = window.evaluate_js("document.getElementById('events').textContent")
        finally:
            window.destroy()

    webview.start(check, gui="edgechromium", http_server=True)
    sink.close()
    expected = len([name for name in dir(api) if not name.startswith("_")])
    passed = seen.get("methods") == expected and "smoke" in (seen.get("events") or "")
    print(seen)
    print("PASS" if passed else f"FAIL (expected {expected} methods and a smoke event)")
    return passed


def check_page() -> bool:
    index = REPO_ROOT / "ui" / "dist" / "index.html"
    if not index.is_file():
        print("FAIL (ui\\dist\\index.html is missing: run `npm run build` in ui\\ first)")
        return False
    api, sink = _api()
    window = webview.create_window("Rosetta smoke test (page)", url=str(index), js_api=api, width=1280, height=800)
    sink.attach(window.evaluate_js)
    seen = {}

    def check():
        try:
            deadline = time.monotonic() + READY_TIMEOUT_S
            while time.monotonic() < deadline and not seen.get("ready"):
                seen["ready"] = window.evaluate_js("window.rosettaReady === true")
                time.sleep(0.25)
            seen["shell"] = window.evaluate_js("document.querySelector('rp-app') !== null")
            seen["text"] = window.evaluate_js("document.body.innerText.slice(0, 120)")
        finally:
            window.destroy()

    webview.start(check, gui="edgechromium", http_server=True)
    sink.close()
    passed = bool(seen.get("ready")) and bool(seen.get("shell"))
    print(seen)
    print("PASS" if passed else "FAIL (the page didn't report itself ready)")
    return passed


if __name__ == "__main__":
    sys.exit(0 if (check_page() if "--page" in sys.argv[1:] else check_console()) else 1)
